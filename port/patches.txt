# Draw distance (--drawdistance N): the race projection's far plane and the
# camera-distance cull for props/effects scale with sbk_far_scale (port_stubs.c).
# configureViewport's 2800 is the race camera, and it also tells the
# Enhanced-mode distance haze (port/src/gfx/haze.c) where the plane landed --
# sbk_haze_note_far returns its argument, so the expression is unchanged for
# the game and the haze gets the one number it cannot work out for itself.
# The other two are not the race camera: (f32)farClip is
# configureViewportWithFovAndFarClip, which only the ending credits use.
src/engine/viewport_manager.c	        2800.0f,	        sbk_haze_note_far(2800.0f * sbk_far_scale),
src/engine/viewport_manager.c	        (f32)farClip,	        (f32)farClip * sbk_far_scale,
#
# The same range is what the Enhanced-mode far-object fade-in (`fadein`) fades
# across: sbk_fadein_note_cull hands the port the number the game is really
# culling at this run, and returns it unchanged.
src/math/spatial_math.c	#define RACE_VIEWPORT_CAMERA_CULL_RANGE 0xBA00000	#define RACE_VIEWPORT_CAMERA_CULL_RANGE sbk_fadein_note_cull((s32)(0xBA00000 * sbk_far_scale))

# Which matrices belong to an object the cull applies to (`fadein`).
# allocFixedTransformMatrix is where the game builds a prop's, an effect's or
# a projectile's transform -- the things isPositionNearCurrentRaceViewportCamera
# decides about -- so the port is handed each pointer and fades only draws whose
# modelview is one of them. Terrain and the sky never appear in the table,
# which is what stops a distant chunk of course being faded like a distant prop
# (the sequel's half of this measured 20,000 triangles a second going to alpha
# 0 before the table existed). See port/src/gfx/haze.c.
src/menu/renderer/menu_render_utils.c	    PACKED_MTX_WORD(dst, 15) = (source->translation.z << 16) & 0xFFFF0000;	    PACKED_MTX_WORD(dst, 15) = (source->translation.z << 16) & 0xFFFF0000;\n    { extern void sbk_fadein_note_object(const void *); sbk_fadein_note_object(dst); }
